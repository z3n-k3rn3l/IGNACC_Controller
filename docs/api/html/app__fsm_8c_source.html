71static void handle_relay_command(app_fsm_t *self, const proto_cmd_t *cmd)
72{

74 unsigned index = cmd->channel - 1u;

134static void handle_command(app_fsm_t *self, const proto_cmd_t *cmd)
135{
136 char buf[PROTO_RESPONSE_MAX];
137
138 switch (cmd->kind) {
139 case PROTO_CMD_RELAY:
140 if (cmd->channel >= 1 && cmd->channel <= relay_bank_count(&self->cfg.relays)) {
141 handle_relay_command(self, cmd);
142 } else {
143 /* A channel this board doesn't have: unknown here. */
144 send_response(self, buf, proto_format_nack(buf, sizeof(buf)));
145 }
146 break;
147 case PROTO_CMD_ALL:
148 handle_all_command(self, cmd);
149 break;
150 case PROTO_CMD_HELLO:
151 send_response(self, buf, proto_format_hello(self->cfg.fw_version, buf, sizeof(buf)));
152 break;
153 case PROTO_CMD_STATUS:
154 handle_status(self);
155 break;
156 case PROTO_CMD_NONE:
157 default:
158 send_response(self, buf, proto_format_nack(buf, sizeof(buf)));
159 break;
160 }
161}

163app_fsm_t *app_fsm_create(const app_fsm_config_t *cfg)
164{
165 if (cfg == NULL || cfg->link.ops == NULL || cfg->fw_version == NULL || cfg->delay == NULL ||
166 cfg->relays.relays == NULL || cfg->relays.count < 1 || cfg->relays.count > PROTO_MAX_CHANNELS) {
167 return NULL;
168 }
169 for (unsigned i = 0; i < cfg->relays.count; i++) {
170 if (cfg->relays.relays[i].ops == NULL) {
171 return NULL;
172 }
173 }
174
175 app_fsm_t *self = malloc(sizeof(*self) + cfg->relays.count * sizeof(relay_t));
176 if (self == NULL) {
177 return NULL;
178 }
179 memcpy(self->relays, cfg->relays.relays, cfg->relays.count * sizeof(relay_t));
180 self->cfg = *cfg;
181 self->cfg.relays.relays = self->relays;
182 self->state = APP_FSM_STATE_INIT;
183 return self;
184}
