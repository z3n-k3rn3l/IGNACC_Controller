61relay_err_t relay_gpio_create(const relay_gpio_config_t *cfg, const pin_hal_t *hal, relay_t *out)
62{
63 if (cfg == NULL || hal == NULL || hal->ops == NULL || out == NULL || cfg->pin < 0 ||
64 (cfg->active_level != 0 && cfg->active_level != 1)) {
65 return RELAY_ERR_ARG;
66 }
67
68 relay_gpio_t *self = malloc(sizeof(*self));
69 if (self == NULL) {
70 return RELAY_ERR_NOMEM;
71 }
72 self->hal = *hal;
73 self->pin = cfg->pin;
74 self->on_level = cfg->active_level;
75 self->off_level = !cfg->active_level;
76
77 /* Latch OFF before the output driver is enabled: no ON glitch at boot. */
78 if (self->hal.ops->configure_output(self->hal.ctx, self->pin, self->off_level) != 0) {
79 free(self);
80 return RELAY_ERR_IO;
81 }
82 if (self->hal.ops->get_level(self->hal.ctx, self->pin) != self->off_level) {
83 free(self);
84 return RELAY_ERR_VERIFY;
85 }
86
87 out->ops = &s_relay_gpio_ops;
88 out->ctx = self;
89 return RELAY_OK;
90}
