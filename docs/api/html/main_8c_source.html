122void app_main(void)
123{
124 /* 1. Relays first, so all are OFF as early as possible. */
125 const pin_hal_t hal = pin_hal_esp();
126 relay_t relays[CONFIG_IGNACC_RELAY_COUNT];
127 for (unsigned i = 0; i < CONFIG_IGNACC_RELAY_COUNT; i++) {
128 const relay_gpio_config_t relay_cfg = {
129 .pin = s_relay_gpios[i],
130 .active_level = CONFIG_IGNACC_RELAY_ACTIVE_LEVEL,
131 };
132 if (relay_gpio_create(&relay_cfg, &hal, &relays[i]) != RELAY_OK) {
133 halt();
134 }
135 }
136 app_fsm_config_t fsm_cfg = {
137 .relays = { .relays = relays, .count = CONFIG_IGNACC_RELAY_COUNT },
138 .fw_version = esp_app_get_description()->version,
139 .settle_ms = CONFIG_IGNACC_RELAY_SETTLE_MS,
140 .delay = app_fsm_rtos_delay_ms,
141 .delay_ctx = NULL,
142 };
143
144 /* 2. Command queue shared by the link (producer) and the FSM (consumer). */
145 QueueHandle_t line_queue = xQueueCreate(CONFIG_IGNACC_CMD_QUEUE_DEPTH, sizeof(link_line_t));
146 if (line_queue == NULL) {
147 halt();
148 }
149
150 /* 3. Link, then state machine. Lines received meanwhile wait in the queue. */
151 const uart_link_config_t link_cfg = {
152 .port = CONFIG_IGNACC_UART_PORT,
153 .baud_rate = CONFIG_IGNACC_UART_BAUD,
154 .tx_pin = CONFIG_IGNACC_UART_TX_GPIO,
155 .rx_pin = CONFIG_IGNACC_UART_RX_GPIO,
156 .rx_buf_size = CONFIG_IGNACC_UART_RX_BUF_SIZE,
157 .event_queue_len = CONFIG_IGNACC_UART_EVENT_QUEUE_LEN,
158 .max_line_len = CONFIG_IGNACC_MAX_LINE_LEN,
159 .task_priority = CONFIG_IGNACC_UART_TASK_PRIORITY,
160 .task_stack_size = CONFIG_IGNACC_UART_TASK_STACK,
161 .line_queue = line_queue,
162 };
163 if (uart_link_start(&link_cfg, &fsm_cfg.link) != ESP_OK) {
164 halt();
165 }
166
167 /* The FSM copies the relay handles, so the local array may go out of scope. */
168 app_fsm_t *fsm = app_fsm_create(&fsm_cfg);
169 if (fsm == NULL || !app_fsm_start(fsm)) {
170 halt();
171 }
172
173 /* 4. Start consuming commands. */
174 const app_fsm_task_config_t task_cfg = {
175 .line_queue = line_queue,
176 .priority = CONFIG_IGNACC_FSM_TASK_PRIORITY,
177 .stack_size = CONFIG_IGNACC_FSM_TASK_STACK,
178 };
179 if (app_fsm_task_start(fsm, &task_cfg) != ESP_OK) {
180 halt();
181 }
182}
