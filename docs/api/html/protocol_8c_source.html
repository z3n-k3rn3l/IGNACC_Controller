55static proto_cmd_t parse_relay(const char *line, size_t len)
56{
57 if (len < 3 || line[0] != 'I' || line[1] != 'N' || line[2] < '1' || line[2] > '9') {
58 return s_none;
59 }
60 size_t i = 2;
61 unsigned channel = 0;
62 while (i < len && line[i] >= '0' && line[i] <= '9') {
63 channel = channel * 10u + (unsigned)(line[i] - '0');
64 if (channel > PROTO_MAX_CHANNELS) {
65 return s_none;
66 }
67 i++;
68 }
69 const char *rest = line + i;
70 size_t rest_len = len - i;
71 if (equals(rest, rest_len, "_ON")) {
72 return (proto_cmd_t){ PROTO_CMD_RELAY, channel, true };
73 }
74 if (equals(rest, rest_len, "_OFF")) {
75 return (proto_cmd_t){ PROTO_CMD_RELAY, channel, false };
76 }
77 return s_none;
78}

107int proto_format_result(const proto_cmd_t *cmd, bool ok, char *buf, size_t size)
108{
109 if (cmd == NULL || buf == NULL || size == 0) {
110 return -1;
111 }
112 const char *state = cmd->on ? "ON" : "OFF";
113 const char *result = ok ? "OK" : "FAIL";
114 int n;
115
116 switch (cmd->kind) {
117 case PROTO_CMD_RELAY:
118 if (cmd->channel < 1 || cmd->channel > PROTO_MAX_CHANNELS) {
119 return -1;
120 }
121 n = snprintf(buf, size, "IN%u_%s_%s\r\n", cmd->channel, state, result);
122 break;
123 case PROTO_CMD_ALL:
124 n = snprintf(buf, size, "ALL_%s_%s\r\n", state, result);
125 break;
126 default:
127 return -1;
128 }
129 return format_checked(size, n);
130}

149int proto_format_status(const bool *on, unsigned count, char *buf, size_t size)
150{
151 if (on == NULL || count < 1 || count > PROTO_MAX_CHANNELS || buf == NULL || size == 0) {
152 return -1;
153 }
154 size_t used = 0;
155 int n = snprintf(buf, size, "STATUS");
156 if (format_checked(size, n) < 0) {
157 return -1;
158 }
159 used = (size_t)n;
160 for (unsigned i = 0; i < count; i++) {
161 n = snprintf(buf + used, size - used, " IN%u=%s", i + 1u, on[i] ? "ON" : "OFF");
162 if (format_checked(size - used, n) < 0) {
163 return -1;
164 }
165 used += (size_t)n;
166 }
167 n = snprintf(buf + used, size - used, "\r\n");
168 if (format_checked(size - used, n) < 0) {
169 return -1;
170 }
171 return (int)(used + (size_t)n);
172}
