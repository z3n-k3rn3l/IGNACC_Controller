60static void uart_link_read_data(uart_link_t *self, size_t available)
61{
62 uint8_t chunk[RX_CHUNK_SIZE];
63 link_line_t line;
64
65 while (available > 0) {
66 size_t want = available < sizeof(chunk) ? available : sizeof(chunk);
67 int n = uart_read_bytes(self->port, chunk, want, 0);
68 if (n <= 0) {
69 return;
70 }
71 available -= (size_t)n;
72 for (int i = 0; i < n; i++) {
73 if (line_asm_push(&self->assembler, (char)chunk[i], &line)) {
74 /* Never block the receiver: if the queue is full the line is dropped. */
75 (void)xQueueSend(self->lines, &line, 0);
76 }
77 }
78 }
79}

90static void uart_link_task(void *arg)
91{
92 uart_link_t *self = arg;
93 uart_event_t event;
94
95 for (;;) {
96 if (xQueueReceive(self->events, &event, portMAX_DELAY) != pdTRUE) {
97 continue;
98 }
99 switch (event.type) {
100 case UART_DATA:
101 uart_link_read_data(self, event.size);
102 break;
103 case UART_FIFO_OVF:
104 case UART_BUFFER_FULL:
105 /* Bytes were lost: drop everything buffered and the line in progress. */
106 uart_flush_input(self->port);
107 xQueueReset(self->events);
108 line_asm_discard(&self->assembler);
109 break;
110 case UART_FRAME_ERR:
111 case UART_PARITY_ERR:
112 line_asm_discard(&self->assembler);
113 break;
114 default:
115 break;
116 }
117 }
118}

120esp_err_t uart_link_start(const uart_link_config_t *cfg, link_tx_t *out_tx)
121{
122 if (cfg == NULL || out_tx == NULL || cfg->line_queue == NULL || cfg->max_line_len < 1 ||
123 cfg->max_line_len > LINK_LINE_MAX_LEN) {
124 return ESP_ERR_INVALID_ARG;
125 }
126
127 uart_link_t *self = calloc(1, sizeof(*self));
128 if (self == NULL) {
129 return ESP_ERR_NO_MEM;
130 }
131 self->port = (uart_port_t)cfg->port;
132 self->lines = cfg->line_queue;
133 line_asm_init(&self->assembler, cfg->max_line_len);
134
135 const uart_config_t uart_cfg = {
136 .baud_rate = cfg->baud_rate,
137 .data_bits = UART_DATA_8_BITS,
138 .parity = UART_PARITY_DISABLE,
139 .stop_bits = UART_STOP_BITS_1,
140 .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
141 .source_clk = UART_SCLK_DEFAULT,
142 };
143
144 esp_err_t err = uart_driver_install(self->port, cfg->rx_buf_size, 0, cfg->event_queue_len,
145 &self->events, 0);
146 if (err != ESP_OK) {
147 free(self);
148 return err;
149 }
150 err = uart_param_config(self->port, &uart_cfg);
151 if (err == ESP_OK) {
152 err = uart_set_pin(self->port, cfg->tx_pin, cfg->rx_pin, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE);
153 }
154 if (err == ESP_OK) {
155 err = uart_flush_input(self->port);
156 }
157 if (err == ESP_OK &&
158 xTaskCreate(uart_link_task, "uart_link", cfg->task_stack_size, self, cfg->task_priority, NULL) != pdPASS) {
159 err = ESP_ERR_NO_MEM;
160 }
161 if (err != ESP_OK) {
162 uart_driver_delete(self->port);
163 free(self);
164 return err;
165 }
166
167 out_tx->ops = &s_uart_link_tx_ops;
168 out_tx->ctx = self;
169 return ESP_OK;
170}
